Production Backend API (Node.js + Android Gradle)

In production, point POST /api/build-apk to a Node.js/Express worker with Android SDK build-tools (aapt2, zipalign, apksigner) to compile and sign the APK on the server: 

// server.js (Express + Gradle APK Builder)
import express from 'express';
import fs from 'fs-extra';
import { execSync } from 'child_process';

const app = express();
app.use(express.json({ limit: '10mb' }));

app.post('/api/build-apk', async (req, res) => {
  const { url, appName, packageName, iconBase64 } = req.body;
  const workDir = `/tmp/builds/${Date.now()}`;
  await fs.copy('./android-webview-template', workDir);

  // 1. Inject URL & App Name into res/values/strings.xml
  const stringsXml = `<resources>
    <string name="app_name">${escapeXml(appName)}</string>
    <string name="target_url">${escapeXml(url)}</string>
  </resources>`;
  await fs.writeFile(`${workDir}/app/src/main/res/values/strings.xml`, stringsXml);

  // 2. Write custom launcher icon
  if (iconBase64) {
    const buf = Buffer.from(iconBase64.split(',')[1], 'base64');
    await fs.writeFile(`${workDir}/app/src/main/res/mipmap-xxxhdpi/ic_launcher.png`, buf);
  }

  // 3. Build & sign APK with Gradle
  execSync(`./gradlew assembleRelease -PappPackage=${packageName}`, { cwd: workDir });
  const apkPath = `${workDir}/app/build/outputs/apk/release/app-release.apk`;
  res.download(apkPath, `${slugify(appName)}.apk`);
});

app.listen(4000, () => console.log('APK Builder API listening on :4000'));
